cmake_minimum_required(VERSION 3.16)
project(radish_ncnn_test C CXX)
set(CMAKE_CXX_STANDARD 11)
add_subdirectory(${NCNN_SOURCE} ncnn)
add_executable(radish-ncnn main.cpp)
target_link_libraries(radish-ncnn PRIVATE ncnn)
target_include_directories(radish-ncnn PRIVATE ${NCNN_SOURCE}/src/layer)
add_executable(radish-operators operators.cpp)
target_link_libraries(radish-operators PRIVATE ncnn)
target_include_directories(radish-operators PRIVATE ${NCNN_SOURCE}/src/layer)
add_executable(radish-model model.cpp radish_snake.cpp radish_affine_snake.cpp radish_norm_affine_snake.cpp)
target_link_libraries(radish-model PRIVATE ncnn)
target_include_directories(radish-model PRIVATE ${NCNN_SOURCE}/src/layer)
add_executable(radish-packed packed.cpp)
target_link_libraries(radish-packed PRIVATE ncnn)
target_include_directories(radish-packed PRIVATE ${NCNN_SOURCE}/src/layer)
add_executable(radish-snake snake_probe.cpp radish_snake.cpp)
target_link_libraries(radish-snake PRIVATE ncnn)
target_include_directories(radish-snake PRIVATE ${NCNN_SOURCE}/src/layer)
target_compile_options(radish-snake PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
add_executable(radish-affine-snake affine_snake_probe.cpp radish_affine_snake.cpp)
target_link_libraries(radish-affine-snake PRIVATE ncnn)
target_include_directories(radish-affine-snake PRIVATE ${NCNN_SOURCE}/src/layer)
target_compile_options(radish-affine-snake PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
add_executable(radish-norm-affine-snake norm_affine_snake_probe.cpp radish_norm_affine_snake.cpp radish_affine_snake.cpp)
target_link_libraries(radish-norm-affine-snake PRIVATE ncnn)
target_include_directories(radish-norm-affine-snake PRIVATE ${NCNN_SOURCE}/src/layer)
target_compile_options(radish-norm-affine-snake PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
set_source_files_properties(radish_norm_affine_snake.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off;-Wall;-Wextra;-Werror")
if(RADISH_ORT_INCLUDE)
    add_executable(radish-kokoro-runtime kokoro_runtime.cpp ../kokoro_frontend.cpp
        ../kokoro_waveform.cpp ../kokoro_shapes.cpp radish_snake.cpp radish_affine_snake.cpp radish_norm_affine_snake.cpp)
    target_link_libraries(radish-kokoro-runtime PRIVATE ncnn dl)
    target_include_directories(radish-kokoro-runtime PRIVATE ${NCNN_SOURCE}/src/layer ${RADISH_ORT_INCLUDE})
    target_compile_definitions(radish-kokoro-runtime PRIVATE RADISH_FRONTEND_NO_MAIN RADISH_WAVEFORM_NO_MAIN)
    target_compile_options(radish-kokoro-runtime PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
endif()

add_executable(radish-gpu-elapsed EXCLUDE_FROM_ALL gpu_elapsed_probe.cpp)
target_link_libraries(radish-gpu-elapsed PRIVATE ncnn)
target_include_directories(radish-gpu-elapsed PRIVATE ${NCNN_SOURCE}/src/layer)
target_compile_options(radish-gpu-elapsed PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
